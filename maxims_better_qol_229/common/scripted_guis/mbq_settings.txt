# Better Quality of Life 0.3.0: switches of the settings window (gui/mbq_settings.gui), one per checkbox.
# Clicking a ticked box removes its state (= feature off). Conversion: Answer automatically and Mute letters are independent since 2.1.2; tribute: both modes independent (0.3.11).
# Pattern: vanilla scripted gui with an effect block (common/scripted_guis/00_religion.txt:128-144); set_global_variable effects.log:2758-2761.
mbq_toggle_divorce = {
	scope = character
	effect = {
		if = { limit = { has_global_variable = mbq_auto_divorce } remove_global_variable = mbq_auto_divorce }
		else = { set_global_variable = { name = mbq_auto_divorce value = yes } }
	}
}

mbq_toggle_conv_auto = {
	scope = character
	effect = {
		if = { limit = { has_global_variable = mbq_silent_conversion } remove_global_variable = mbq_silent_conversion }
		else = { set_global_variable = { name = mbq_silent_conversion value = yes }
			mbq_conv_prio_defaults_effect = yes # 0.3.22: first start and old saves: empty slots get weak hook, strong hook, favor, nothing
		}
	}
}

# Better Quality of Life 2.1.35 (Maxim 25-10-2026): Pfeile bei Bekehrung automatisch verlangen, Prioritaet 1 bis 4: mbq_prio_swap_N_M tauscht Prioritaet N mit M.
# Nur wenn 'Automatisch antworten' (mbq_silent_conversion) an ist, sonst sind die Prioritaeten gesperrt (MBQ_LOCK_PRIO).

	scope = character
	effect = {
		if = {
			limit = { has_global_variable = mbq_silent_conversion }
			mbq_prio_swap_effect = { A = 1 B = 2 }
		}
	}
}

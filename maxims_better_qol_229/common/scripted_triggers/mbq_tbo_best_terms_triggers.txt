# Better Quality of Life 0.3.7 (Best terms): can at least one ticked box still change the new tributary's contract?
# Scope: the new tributary. Same contract choice as common/scripted_effects/mbq_best_terms.txt; triggers.log:7776, :7781.
mbq_tbo_any_change_trigger = {
	OR = {
		AND = {
			mbq_tbo_want_tax_trigger = yes
			OR = {
				AND = { has_subject_contract_group = tributary_nomadic tributary_contract_obligation_level_can_be_increased = nomad_government_herd }
				AND = { NOT = { has_subject_contract_group = tributary_nomadic } tributary_contract_obligation_level_can_be_increased = default_tributary_taxes }
			}
		}
		AND = {
			mbq_tbo_want_prestige_trigger = yes
			OR = {
				AND = { OR = { has_subject_contract_group = tributary_nomadic has_subject_contract_group = tributary_steppe } tributary_contract_obligation_level_can_be_increased = nomad_government_prestige }
				AND = { NOR = { has_subject_contract_group = tributary_nomadic has_subject_contract_group = tributary_steppe } tributary_contract_obligation_level_can_be_increased = default_tributary_prestige }
			}
		}
		AND = {
			mbq_tbo_want_war_support_trigger = yes
			tributary_contract_obligation_level_can_be_increased = tributary_war_participation_obligation
		}
		AND = {
			mbq_tbo_want_no_guarantee_trigger = yes
			tributary_contract_obligation_level_can_be_decreased = suzerain_war_participation_guarantee
		}
	}
}

mbq_tbo_want_tax_trigger = { has_global_variable = mbq_tbo_tax }
mbq_tbo_want_prestige_trigger = { has_global_variable = mbq_tbo_prestige }
mbq_tbo_want_war_support_trigger = { has_global_variable = mbq_tbo_war_support }
mbq_tbo_want_no_guarantee_trigger = { NOT = { has_global_variable = mbq_tbo_no_guar_off } }

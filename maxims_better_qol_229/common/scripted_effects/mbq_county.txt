# Die eigentliche Vergabe (bis 2.0.14 mbq_gr_do_grant_effect). Nur ueber mbq_gr_do_grant_effect aufrufen.
mbq_gr_do_grant_checked_effect = {
	# Stress fuer gierig/ehrgeizig 1:1 GT:1020-1080 (Liste target_titles = der eine Titel). Vor der Uebergabe geprueft = Stand beim Entschluss;
	# ob Vanilla vor oder nach dem Code-Transfer prueft, ist unbelegt (GT:1). Greift nur an/unter der Schwelle und nur bei Grafschaft/Baronie.
	if = {
		limit = {
			OR = {
				has_trait = greedy
				has_trait = ambitious
			}
			domain_size <= domain_limit
			scope:mbq_gr_grant_title.tier <= tier_county
		}
		if = {
			limit = { domain_size = domain_limit }
			stress_and_fulfillment_impact = { greedy = minor_stress_impact_gain ambitious = minor_stress_impact_gain } # GT:1061-1069
		}
		else_if = {
			limit = { domain_size < domain_limit }
			stress_and_fulfillment_impact = { greedy = medium_stress_impact_gain ambitious = medium_stress_impact_gain } # GT:1070-1078
		}
	}
	scope:mbq_gr_grant_title = {
		create_title_and_vassal_change = { type = granted save_scope_as = mbq_gr_change }
		if = { # Logikcheck M4: government_base Pflicht bei unbelandetem Empfaenger (E:177-179), Regierung des Spielers wie Vanilla-Verleihen
			limit = { scope:mbq_gr_recipient = { is_landed = no } }
			change_title_holder_include_vassals = {
				holder = scope:mbq_gr_recipient
				change = scope:mbq_gr_change
				take_baronies = no
				government_base = scope:mbq_gr_liege
			}
		}
		else = {
			change_title_holder_include_vassals = {
				holder = scope:mbq_gr_recipient
				change = scope:mbq_gr_change
				take_baronies = no
			}
		}
		resolve_title_and_vassal_change = scope:mbq_gr_change
	}
	scope:mbq_gr_recipient = {
		if = {
			limit = { scope:mbq_gr_grant_title.tier = tier_barony }
			add_opinion = { target = scope:mbq_gr_liege modifier = received_title_barony }
		}
		else_if = {
			limit = { scope:mbq_gr_grant_title.tier = tier_county }
			add_opinion = { target = scope:mbq_gr_liege modifier = received_title_county }
		}
		# Hoefische Vasallen gegen Niedriggeborene 1:1 GT:575-590 (Spieler-Zweig, Grafschaft -5), Baronie nichts (GT:551-560); Herzogtum/Koenigreich gibt die Mod ab 2.1 nicht mehr ab
		if = {
			limit = {
				is_lowborn = yes
				NOT = { scope:mbq_gr_grant_title.tier = tier_barony }
			}
			scope:mbq_gr_liege = {
				if = {
					limit = { scope:mbq_gr_grant_title.tier = tier_county }
					every_vassal = {
						vassal_stance = courtly
						add_opinion = { target = scope:mbq_gr_liege modifier = courtly_lowborn_grant_opinion opinion = -5 }
					}
				}
			}
		}
	}
	if = { # Empfaenger-Bau: neuer Adliger wie Vanilla (Wirkung der on_actions, nur fuer den Titel, der ihn erzeugt hat)
		limit = { exists = scope:mbq_gr_new_noble scope:mbq_gr_new_noble > 0 } # 2.0.16
		mbq_gr_new_noble_after_effect = yes
		scope:mbq_gr_recipient = { # Logikcheck 20 S2: Liste wird nur einmal je Lauf gefuellt; der neue Direktvasall bleibt fuer die naechsten Titel Kandidat wie bisher
			if = { limit = { NOT = { is_in_list = mbq_gr_cands } } add_to_temporary_list = mbq_gr_cands }
		}
		save_scope_value_as = { name = mbq_gr_new_noble value = 0 } # 2.0.16: Ueberschreiben statt clear_saved_scope
	}
	# Sammelmeldung statt Toast je Titel (MENUE_RICHTLINIE 11); Stumm-Modus: nichts merken, keine Meldung
	if = {
		limit = { NOT = { has_global_variable = mbq_gr_silent } }
		if = { # Titel nur einmal merken (Doppelmeldung 14-10; triggers.log:815 is_target_in_variable_list)
			limit = { NOT = { is_target_in_variable_list = { name = mbq_gr_list target = scope:mbq_gr_grant_title } } }
			add_to_variable_list = { name = mbq_gr_list target = scope:mbq_gr_grant_title } # E:58
		}
		mbq_gr_pending_effect = yes
	}
}

# Kandidaten (Trichter-Bau 14-10): je Titel sammeln (nur Neue kommen dazu), dann nach Filter und Rang waehlen (Leistung: nur Reich, nur beim Lauf).
# Temporaere Liste lebt "later in the same effect" (effects.log:83); Doppelte
# verhindert is_in_list (triggers.log:780, Form Better Marriages mmoc_scripted_effects.txt:112). Wer: Eure Hoeflinge (E:4499), Direktvasallen
# (E:5360; Option A: keine Untervasallen), lebende Mitglieder Eures Hauses (E:13502), Nachkommen bis Ururenkel (E:4398), Dynastie (E:16178) nur, wenn ein Platz
# Dynasty waehlt (alte Dynastien zaehlen Tote mit, empfaenger_konzept.md H2). Familienschleifen filtern im limit mbq_gr_realm_member_trigger
# (nur Reich, unbelandet oder Direktvasall); so kommen Verwandte an Vasallenhoefen herein, fremde Hoeflinge dort nie. Ein eigenes
# every_vassal { every_courtier } waere doppelt: Haus-, Kinder- und Dynastieschleife finden dieselben Leute.
# Im Reich bleiben nur Gueltige: mbq_gr_recipient_valid_trigger (target_is_liege_or_above) im Kandidaten-Filter. Scope = Spieler.
mbq_gr_cands_effect = {
	if = { # Logikcheck 20 S2: einmal je Lauf fuellen. Merker = der Spieler selbst in derselben temporaeren Liste (lebt genau so lange wie die Liste,
		# effects.log:83; ein Scope-Wert koennte eine Event-Option ueberleben, die Liste nicht). Der Spieler faellt im Kandidaten-Filter raus
		# (mbq_gr_cand_ok_trigger NOT = { this = scope:mbq_gr_liege }). Titelabhaengig ist nur der Filter/Rang (ordered_in_list je Titel), nicht die Sammlung.
		limit = { scope:mbq_gr_liege = { NOT = { is_in_list = mbq_gr_cands } } }
		scope:mbq_gr_liege = { add_to_temporary_list = mbq_gr_cands }
		scope:mbq_gr_liege = {
			every_courtier = {
				limit = { NOT = { is_in_list = mbq_gr_cands } }
				add_to_temporary_list = mbq_gr_cands
			}
			every_vassal = { # nur DIREKTE Vasallen (E:5360), keine Untervasallen (Option A, konzept.md 10)
				limit = { NOT = { is_in_list = mbq_gr_cands } }
				add_to_temporary_list = mbq_gr_cands
			}
			house ?= {
				every_house_member = {
					limit = {
						is_alive = yes
						NOT = { is_in_list = mbq_gr_cands }
						mbq_gr_realm_member_trigger = yes
					}
					add_to_temporary_list = mbq_gr_cands
				}
			}
			every_child = {
				if = { limit = { is_alive = yes NOT = { is_in_list = mbq_gr_cands } mbq_gr_realm_member_trigger = yes } add_to_temporary_list = mbq_gr_cands }
				every_child = {
					if = { limit = { is_alive = yes NOT = { is_in_list = mbq_gr_cands } mbq_gr_realm_member_trigger = yes } add_to_temporary_list = mbq_gr_cands }
					every_child = {
						if = { limit = { is_alive = yes NOT = { is_in_list = mbq_gr_cands } mbq_gr_realm_member_trigger = yes } add_to_temporary_list = mbq_gr_cands }
						every_child = {
							limit = {
								is_alive = yes
								NOT = { is_in_list = mbq_gr_cands }
								mbq_gr_realm_member_trigger = yes
							}
							add_to_temporary_list = mbq_gr_cands
						}
					}
				}
			}
			if = {
				limit = { mbq_gr_uses_dynasty_trigger = yes }
				dynasty ?= {
					every_dynasty_member = {
						limit = {
							is_alive = yes
							NOT = { is_in_list = mbq_gr_cands }
							mbq_gr_realm_member_trigger = yes
						}
						add_to_temporary_list = mbq_gr_cands
					}
				}
			}
		}
	}
}

# Niemand passt (Maxim 14-10): der Titel bleibt, die Sammelmeldung nennt den Grund. Je Titel hoechstens einmal in 360 Tagen gemeldet, sonst kaeme
# bei jedem Takt dieselbe Zeile. Scope = Spieler, scope:mbq_gr_grant_title = Titel.
mbq_gr_nofit_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = mbq_gr_silent }
			scope:mbq_gr_grant_title = { NOT = { has_variable = mbq_gr_nofit } }
		}
		scope:mbq_gr_grant_title = { set_variable = { name = mbq_gr_nofit value = yes days = 360 } }
		if = {
			limit = { has_variable = mbq_gr_kept_n }
			change_variable = { name = mbq_gr_kept_n add = 1 }
		}
		else = {
			set_variable = { name = mbq_gr_kept_n value = 1 }
		}
		if = { # 2.2.8: die Vasallenschwelle wird in der Sammelmeldung nur genannt, wenn sie fuer DIESEN Titel gilt und voll ist (neue Direktvasallen und Rueckfall ausgeschlossen); Baronien nie
			limit = { mbq_gr_vassal_slot_ok_trigger = no }
			if = {
				limit = { has_variable = mbq_gr_kept_full }
				change_variable = { name = mbq_gr_kept_full add = 1 }
			}
			else = {
				set_variable = { name = mbq_gr_kept_full value = 1 }
			}
		}
		mbq_gr_pending_effect = yes
	}
}

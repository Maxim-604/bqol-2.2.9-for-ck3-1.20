# FULL COPIES of vanilla CK3 1.20 interactions (database override, the later definition wins):
#   divorce_character_rel_head_interaction                  = 00_marriage_interactions.txt:4385-5450
#   divorce_character_dynast_request_rel_head_interaction   = 00_marriage_interactions.txt:5687-6721
# Only change: the auto_accept block (between 'mbq PATCH START' and 'mbq PATCH END'),
# vanilla 00_marriage_interactions.txt:5031-5038 and :6296-6303.

divorce_character_rel_head_interaction = {
	category = interaction_category_diplomacy
	use_diplomatic_range = no
	icon = icon_marriage

	cost = {
		piety = {
			value = divorce_cost
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	redirect = {
		scope:recipient = {
			save_scope_as = secondary_recipient
		}	
		#Recipient should be cleric or hof
		if = {
			limit = {
				scope:puppet_or_actor = {
					government_has_flag = government_is_theocracy
					exists = superior
				}
			}
			scope:puppet_or_actor.superior = { save_scope_as = recipient }
		}
		#Redirect to bishop superior of court chaplain
		else_if = {
			limit = {
				OR = {
					scope:puppet_or_actor.highest_held_title_tier < tier_kingdom
					scope:puppet_or_actor.faith = {
						NOR = {
							has_doctrine = doctrine_spiritual_head
							exists = scope:faith.religious_head
						}
					}
				}
				scope:puppet_or_actor.cp:councillor_court_chaplain ?= {
					highest_held_title_tier < tier_duchy
					superior ?= {
						highest_held_title_tier >= tier_duchy
						is_clergy = yes
					}
				}
			}
			scope:puppet_or_actor.cp:councillor_court_chaplain.superior = {
				save_scope_as = recipient
			}
		}
		#Redirect to bishop
		else_if = {
			limit = {
				OR = {
					scope:puppet_or_actor.highest_held_title_tier < tier_kingdom
					scope:puppet_or_actor.faith = {
						NOR = {
							has_doctrine = doctrine_spiritual_head
							exists = scope:faith.religious_head
						}
					}
				}
				scope:puppet_or_actor.cp:councillor_court_chaplain ?= {
					highest_held_title_tier >= tier_duchy
					is_clergy = yes
				}
			}
			scope:puppet_or_actor.cp:councillor_court_chaplain = {
				save_scope_as = recipient
			}
		}
		else = {
			scope:puppet_or_actor.faith.religious_head ?= {
				save_scope_as = recipient
			}
		}
	}

	greeting = positive
	notification_text = {
		desc = ASK_FOR_DIVORCE_NOTIFICATION
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = {
						faith = scope:recipient.faith
						sex_same_as = scope:secondary_recipient
						allowed_to_marry_same_sex_trigger = no
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_SAME_SEX
			}
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = {
						relation_with_character_is_incestuous_in_my_faith_trigger = { CHARACTER = scope:secondary_recipient }
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_INCESTUOUS
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient = {
						any_relation = {
							type = lover
							NOR = {
								this = scope:puppet_or_actor
								this = scope:recipient
							}
							relation_with_character_is_incestuous_in_my_rite_trigger = { CHARACTER = scope:secondary_recipient }
						}
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_INCESTUOUS_LOVER
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient = {
						is_female = no
						is_eunuch_trigger = yes
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_EUNUCH
			}
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = {
						has_opinion_modifier = {
							modifier = attempted_murder_me_crime
							target = scope:secondary_recipient
						}
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_ATTEMPTED_MURDER
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient = {
						has_character_modifier = wedding_accused_impotency_modifier
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_IMPOTENCE
			}
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = {
						any_consort = {
							is_alive = yes
							this != scope:secondary_recipient
						}
						any_consort = {
							this = scope:secondary_recipient
						}
					}
					scope:secondary_recipient = {
						culture = { NOT = { has_cultural_tradition = tradition_polygamous } }
						rite = { NOT = { rite_has_doctrine = doctrine_polygamy } }
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_MULTIPLE_CONSORTS
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient.faith = {
						faith_hostility_level = {
							target = scope:recipient.faith
							value >= faith_hostile_level
						}
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_HOSTILE_FAITH
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient = {
						exposed_cheating_on_spouse_trigger = { SPOUSE = scope:puppet_or_actor }
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_EXPOSED_CHEATING
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient = {
						fertility < 0.1
						NOT = {
							any_child = {
								any_parent = {
									this = scope:puppet_or_actor
								}
							}
						}
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_INFERTILITY
			}
			triggered_desc = {
				trigger = {
					scope:secondary_recipient = {
						has_trait = adulterer
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_ADULTERY_TRAIT
			}
			triggered_desc = {
				trigger = {
					scope:puppet_or_actor = {
						opinion = {
							target = scope:secondary_recipient
							value < -50
						}
					}
				}
				desc = ASK_FOR_DIVORCE_NOTIFICATION_HATRED
			}
			desc = ASK_FOR_DIVORCE_NOTIFICATION_NOT_JUSTIFIED
		}
	}

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			NOT = { has_character_modifier = rekindled_marriage_modifier }
		}
	}

	is_shown = {
		scope:puppet_or_actor.rite ?= { rite_has_doctrine = doctrine_divorce_approval }
		# A clerical ruler grants their own divorce and never petitions for one, least of all themselves
		scope:puppet_or_actor ?= { pam_grants_own_divorce_trigger = no }
		scope:puppet_or_actor ?= {
			OR = {
				faith ?= {
					has_doctrine = doctrine_spiritual_head
					exists = religious_head
				}
				cp:councillor_court_chaplain ?= {
					highest_held_title_tier >= tier_duchy
					is_clergy = yes
				}
				AND = {
					government_has_flag = government_is_theocracy
					exists = superior
				}
				cp:councillor_court_chaplain.superior ?= {
					highest_held_title_tier >= tier_duchy
					is_clergy = yes
				}
			}
		}
		scope:secondary_recipient ?= {
			is_spouse_of = scope:puppet_or_actor
			is_concubine = no # Can set aside Concubines at any time
		}
	}

	is_valid_showing_failures_only = {
		scope:puppet_or_actor = {
			NOT = { has_trait = excommunicated }
		}
		trigger_if = {
			limit = { scope:actor = scope:puppet_or_actor }
			custom_tooltip = {
				NOT = {
					scope:puppet_or_actor.culture = {
						has_cultural_parameter = cannot_divorce
					}
				}
				text = CULTURE_FORBIDS_DIVORCE
			}
		}
		trigger_else = {
			custom_tooltip = {
				NOT = {
					scope:puppet_or_actor.culture = {
						has_cultural_parameter = cannot_divorce
					}
				}
				text = CULTURE_FORBIDS_DIVORCE_puppeteer
			}
		}
		NOT = {
			scope:secondary_recipient = { has_strong_hook = scope:puppet_or_actor }
		}
		scope:puppet_or_actor = {
			trigger_if = {
				limit = { scope:actor = scope:puppet_or_actor }
				custom_tooltip = {
					NOT = {
						var:family_widow_forbidden_divorce ?= scope:secondary_recipient
					}
					text = BOUND_TO_FAMILY_WIDOW
				}
			}
			trigger_else = {
				custom_tooltip = {
					NOT = {
						var:family_widow_forbidden_divorce ?= scope:secondary_recipient
					}
					text = BOUND_TO_FAMILY_WIDOW_puppeteer
				}
			}
		}
		scope:puppet_or_actor = {
			NOT = {
				has_character_modifier = promised_pilgrimage_to_clergy_modifier
			}
		}
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				piety >= divorce_cost
			}
		}
	}

				exists = house

				house = scope:secondary_recipient.house
				NOR = {
					house = scope:puppet_or_actor.house
					is_close_or_extended_family_of = scope:puppet_or_actor

			add = 10
			desc = YOU_ARE_MY_HOUSE_MEMBER
			scope:recipient = {
				is_lowborn = no
				exists = house
				exists = scope:puppet_or_actor.house
				house = scope:puppet_or_actor.house
				NOR = {
					is_close_or_extended_family_of = scope:puppet_or_actor
					is_cousin_of = scope:puppet_or_actor
					is_child_of = scope:puppet_or_actor
					scope:recipient.house = scope:secondary_recipient.house
				}
			}
		}

		modifier = { # Related to the Head of Faith
			add = 25
			desc = YOU_ARE_MY_RELATIVE
			scope:recipient = {
				OR = {
					is_close_or_extended_family_of = scope:puppet_or_actor
					is_cousin_of = scope:puppet_or_actor

		}

		modifier = { # Parent of the Head of Faith

		modifier = {

			add = 30

				has_character_modifier = wedding_accused_impotency_modifier
			}
		}
		# Unity modifiers
		evaluate_action_decreasing_house_unity = {
			VALUE = 100

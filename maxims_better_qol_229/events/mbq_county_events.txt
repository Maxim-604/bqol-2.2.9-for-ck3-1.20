	}
	right_portrait = { # wie mbq_gr.0002 (V events/activities/activity_system_events.txt:90)
		trigger = { exists = scope:mbq_gr_pick_ok scope:mbq_gr_pick_ok = 1 } # 2.0.16
		character = scope:mbq_gr_recipient
		animation = happiness

		save_scope_value_as = { name = mbq_gr_preview value = 1 } # Auswahl wie die Automatik (mbq_gr_county_pick_effect), ohne neuen Adligen zu erschaffen
		mbq_gr_county_pick_effect = yes
		save_scope_value_as = { name = mbq_gr_preview value = 0 } # 2.0.16: Ueberschreiben statt Loeschen (wirkte im Lauf nicht)

			if = {
				limit = { has_variable = mbq_gr_asked }

				limit = { holder ?= root }

			}
		}
	}

			exists = scope:mbq_gr_pick_ok scope:mbq_gr_pick_ok = 1 # 2.0.16

		}
		custom_tooltip = mbq_gr.0001.give_other_tt
		hidden_effect = {
			if = {
				limit = {
					scope:mbq_gr_grant_title.holder ?= root
					mbq_gr_over_value > 0
				}
				mbq_gr_county_grant_effect = yes # neue Auswahl (:1226), sonst nofit bzw. neuer Adliger nach Haken
			}
		}
	}
	option = { # Weitergeben an einen neuen Adligen (Haken New noble if nobody fits, niemand passt)
		name = mbq_gr.0001.give
		trigger = {
			scope:mbq_gr_pick_ok = 0 # 2.0.16

				}
				mbq_gr_county_grant_effect = yes # erschafft jetzt den neuen Adligen (Auswahl ohne Vorschau)
			}
		}
	}
}

# Takt-Schleife (Muster mbq_vassal.0001)
mbq_gr.0010 = {
	hidden = yes
	immediate = {
		if = {
			limit = {
				exists = scope:mbq_gr_gen
				has_variable = mbq_gr_gen
				scope:mbq_gr_gen = var:mbq_gr_gen
			}
			remove_variable = mbq_gr_armed
			if = {
				limit = {
					has_global_variable = mbq_gr_auto
					is_alive = yes
					is_ai = no
				}
				if = {
					limit = { is_imprisoned = no } # Pause in Haft wie Loesegeld und Vassals
					mbq_gr_run_effect = yes
				}
				mbq_gr_arm_effect = yes
			}
		}
		else_if = {
			limit = {
				NOT = { exists = scope:mbq_gr_gen }
				has_global_variable = mbq_gr_auto
				is_alive = yes
				is_ai = no
			}
			mbq_gr_arm_effect = yes
		}
	}
}

# 2.1.9 Wertvoll nach Erbfolge/Teilung (Dev-Test 22-10): der Gewinn-Haken mbq_gr_on_title_gain_kept (mbq_county_on_actions.txt) schreibt markierte Titel, die bei Erbfolge ankommen, in die globale Liste

# Geprueft werden nur die Titel der Liste; danach wird sie geleert (Muster clear_global_variable_list V game_start.txt:7320, every_in_global_list V activity_invite_rules.txt:712).
mbq_gr.9101 = {
	hidden = yes
	immediate = {
		every_in_global_list = {
			variable = mbq_wv_pending
			limit = {
				has_variable = mbq_gr_kept
				NOT = { holder ?= { is_ai = no } }
			}
			remove_variable = mbq_gr_kept
		}
		clear_global_variable_list = mbq_wv_pending
	}
}

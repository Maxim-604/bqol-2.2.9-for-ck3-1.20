	scope = character
	effect = {
		if = { limit = { has_global_variable = mbq_vs_acc } remove_global_variable = mbq_vs_acc }
		else = { set_global_variable = { name = mbq_vs_acc value = yes } }
		mbq_vs_preview_light_effect = yes

}

mbq_vs_toggle_acc_tb = { # Spiegel (Maxim 15-10): Hauptschalter "Accept tributary offers" = mindestens eine der alten Variablen (basis mbq_settings.txt:18-30);
	scope = character    # an setzt beide Gruppen, aus nimmt beide raus. Anders als basis :21/:28 bleiben die Vertragswuensche stehen (gemeinsamer Satz).
	effect = {
		if = {
			limit = { OR = { has_global_variable = mbq_auto_tribute has_global_variable = mbq_auto_tribute_all } }
			if = { limit = { has_global_variable = mbq_auto_tribute } remove_global_variable = mbq_auto_tribute }
			if = { limit = { has_global_variable = mbq_auto_tribute_all } remove_global_variable = mbq_auto_tribute_all }
		}
		else = {
			set_global_variable = { name = mbq_auto_tribute value = yes }
			set_global_variable = { name = mbq_auto_tribute_all value = yes }
		}
		mbq_vs_preview_light_effect = yes # Logikcheck 20 S3
	}
}

mbq_vs_toggle_master = { # 2.1.2 Hauptschalter "Automatik an"
	scope = character
	effect = {
		mbq_menu_migrate_effect = yes # Altspielstand zuerst uebernehmen, sonst wuerde das Umlegen den Zustand der alten Schalter ueberschreiben
		if = { limit = { has_global_variable = mbq_vs_master } remove_global_variable = mbq_vs_master }
		else = {
			set_global_variable = { name = mbq_vs_master value = yes }
			if = {
				limit = {
					OR = { has_global_variable = mbq_vs_auto has_global_variable = mbq_tb_auto }
					NOT = { has_variable = mbq_vs_armed }
				}
				mbq_vs_arm_effect = yes # laufende Schleife nicht verschieben
			}
		}
		mbq_vs_preview_light_effect = yes
	}
}

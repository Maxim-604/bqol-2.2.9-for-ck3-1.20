mbq_dc_arm_effect = {
	if = {
		limit = { has_variable = mbq_dc_gen }
		change_variable = { name = mbq_dc_gen add = 1 }
	}
	else = {
		set_variable = { name = mbq_dc_gen value = 1 }
	}
	save_scope_value_as = { name = mbq_dc_gen value = var:mbq_dc_gen }
	if = {
		limit = { has_global_variable = mbq_dc_t1 }
		set_variable = { name = mbq_dc_armed value = yes days = 35 }
		trigger_event = { id = mbq_dconv.0001 days = 30 }
	}
	else_if = {
		limit = { has_global_variable = mbq_dc_t6 }
		set_variable = { name = mbq_dc_armed value = yes days = 187 }
		trigger_event = { id = mbq_dconv.0001 days = 182 }
	}
	else_if = {
		limit = { has_global_variable = mbq_dc_t12 }
		set_variable = { name = mbq_dc_armed value = yes days = 370 }
		trigger_event = { id = mbq_dconv.0001 days = 365 }
	}
	else = {
		set_variable = { name = mbq_dc_armed value = yes days = 96 }
		trigger_event = { id = mbq_dconv.0001 days = 91 }
	}
}

# Sammelmeldung (aus mbq_dconv.0002): Erfolg = jetzt gleicher Glaube wie der Spieler
mbq_dc_summary_effect = {
	set_variable = { name = mbq_dc_n value = 0 }
	set_variable = { name = mbq_dc_slot value = 0 }
	set_variable = { name = mbq_dc_rn value = 0 } # 2.1.5: zu Eurem Ritus gewechselt (gleicher Glaube, jetzt gleicher Ritus)
	if = { limit = { has_variable_list = mbq_dc_rlist }
		every_in_list = {
			variable = mbq_dc_rlist
			limit = { is_alive = yes faith = root.faith rite = root.rite }
			root = { change_variable = { name = mbq_dc_rn add = 1 } }
		}
		every_in_list = { # 2.1.12: no flag stays behind (answers are in after 5 days, the summary comes after 10)
			variable = mbq_dc_rlist
			limit = { has_variable = mbq_rite_auto }
			remove_variable = mbq_rite_auto
		}
		clear_variable_list = mbq_dc_rlist
	}
	if = { limit = { has_variable_list = mbq_dc_list }
		every_in_list = {
			variable = mbq_dc_list
			limit = { is_alive = yes faith = root.faith }
			root = { change_variable = { name = mbq_dc_n add = 1 } }
		}
		ordered_in_list = {
			variable = mbq_dc_list
			limit = { is_alive = yes faith = root.faith }
			order_by = mbq_vs_order_value # hoechster Rang zuerst (mbq_vassal_values.txt)
			max = 5
			check_range_bounds = no
			save_scope_as = mbq_dc_pr
			root = {
				change_variable = { name = mbq_dc_slot add = 1 }
				if = { limit = { var:mbq_dc_slot = 1 } set_variable = { name = mbq_dc_c1 value = scope:mbq_dc_pr } }
				else_if = { limit = { var:mbq_dc_slot = 2 } set_variable = { name = mbq_dc_c2 value = scope:mbq_dc_pr } }
				else_if = { limit = { var:mbq_dc_slot = 3 } set_variable = { name = mbq_dc_c3 value = scope:mbq_dc_pr } }
				else_if = { limit = { var:mbq_dc_slot = 4 } set_variable = { name = mbq_dc_c4 value = scope:mbq_dc_pr } }
				else_if = { limit = { var:mbq_dc_slot = 5 } set_variable = { name = mbq_dc_c5 value = scope:mbq_dc_pr } }
			}
		}
		clear_variable_list = mbq_dc_list
	}
	if = {
		limit = { var:mbq_dc_n > 5 }
		set_variable = { name = mbq_dc_rest value = var:mbq_dc_n }
		change_variable = { name = mbq_dc_rest subtract = 5 }
	}
	if = {
		limit = {
			OR = {
				var:mbq_dc_n >= 1
				var:mbq_dc_rn >= 1
				AND = { exists = var:mbq_dc_rr var:mbq_dc_rr >= 1 } # 2.1.12
				has_variable = mbq_dc_manual
			}
		}
		send_interface_message = {
			type = mbq_dconv_summary_message
			title = mbq_dc_sum_title
			desc = mbq_dc_sum_desc
		}
	}
	if = { limit = { has_variable = mbq_dc_manual } remove_variable = mbq_dc_manual }
	if = { limit = { has_variable = mbq_dc_pending } remove_variable = mbq_dc_pending }
	remove_variable = mbq_dc_n
	remove_variable = mbq_dc_rn
	if = { limit = { has_variable = mbq_dc_rr } remove_variable = mbq_dc_rr }
	remove_variable = mbq_dc_slot
	if = { limit = { has_variable = mbq_dc_rest } remove_variable = mbq_dc_rest }
	if = { limit = { has_variable = mbq_dc_c1 } remove_variable = mbq_dc_c1 }
	if = { limit = { has_variable = mbq_dc_c2 } remove_variable = mbq_dc_c2 }
	if = { limit = { has_variable = mbq_dc_c3 } remove_variable = mbq_dc_c3 }
	if = { limit = { has_variable = mbq_dc_c4 } remove_variable = mbq_dc_c4 }
	if = { limit = { has_variable = mbq_dc_c5 } remove_variable = mbq_dc_c5 }
}

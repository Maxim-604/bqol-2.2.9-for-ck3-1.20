	}

# 2.1.19 (Maxim 24-10): Startgrafschaften. Alle Grafschaften, die ein Spieler zu Beginn eines NEUEN Spiels haelt, bekommen den Haken Wertvoll und sind damit vor "Titel abgeben" geschuetzt.
# Speicherung des Hakens: Variable mbq_gr_kept am Titel (gui/mbq_wertvoll.gui:28 liest sie, scripted_guis/mbq_county.txt:2149-2174 setzt sie; bei einer Grafschaft auf der Grafschaft selbst,
# deren Baronien sind ueber mbq_county_triggers.txt:110-113 mit geschuetzt und zeigen den Haken). Nur Grafschaften (tier_county), nur wo noch nicht gesetzt; einzeln gehaltene Baronien bleiben unmarkiert.
# Aufruf aus on_game_start_after_lobby (mbq_new_game_on_actions.txt): V common/on_action/game_start.txt:2552-2553 "called once the host (or player, in single player) exits the lobby. Good for
# anything where you need to know who the players are", Beispiel dort :3839 random_player, :3883 every_player; im on_game_start sind die Spieler nicht belegt bekannt. Einmal-Merker
# mbq_start_kept_done (global) und years_from_game_start < 1 (V triggers.log:1097): ein bestehender Spielstand wird nicht angefasst, auch wenn der on_action beim Laden feuern sollte (nicht belegt).
# Verlust: die Markierung faellt bei jedem Titelverlust (mbq_county_on_actions.txt:205-236). Auch von Dev Tools 'Run Rutsch' (Spielstandkopie) gerufen.
mbq_new_game_kept_effect = {
	if = {
		limit = {
			NOT = { has_global_variable = mbq_start_kept_done }
			OR = {
				years_from_game_start < 1
				has_global_variable = mbq_dbg_start_kept # Testhaken nur fuer Dev Tools Run Rutsch (Spielstandkopie ist aelter als ein Jahr); im normalen Spiel nie gesetzt
			}
		}
		set_global_variable = { name = mbq_start_kept_done value = yes }
		if = { limit = { always = no } set_global_variable = { name = mbq_dbg_start_kept value = yes } } # nur ein Setter fuer die Engine-Pruefung (sonst error.log 'is used but is never set', wie mmoc_scripted_effects.txt:504-506); gesetzt wird der Testhaken nur von Dev Tools Run Rutsch
		every_player = {
			set_variable = { name = mbq_kept_seen value = yes } # 2.1.22: dieser Charakter gilt als bearbeitet (kein erneutes Markieren beim Rueckwechsel)
			every_held_title = { # 2.1.46 (Maxim 25-10): jede gehaltene Grafschaft UND jede gehaltene Baronie einzeln (keine Vererbung mehr)
				limit = {
					OR = { tier = tier_county tier = tier_barony }
					NOT = { has_variable = mbq_gr_kept }
				}
				set_variable = { name = mbq_gr_kept value = yes }
			}
			set_variable = { name = mbq_kept_mig_2146 value = yes } # 2.1.46: ein neu markierter Spieler braucht die Umstellung nicht
		}
	}
}

# 2.1.22 (Maxim 24-10): manueller Spielerwechsel. Scope = der NEUE Spielercharakter. Pro Charakter einmal (Variable mbq_kept_seen, auch von mbq_new_game_kept_effect gesetzt), markiert seine gehaltenen
# Grafschaften (tier_county) mit Wertvoll (mbq_gr_kept), nur wo noch keine Markierung ist. Aufruf aus on_player_character_change (mbq_new_game_on_actions.txt) und Dev Tools 'Run Rutsch' (Testfall A7).
mbq_kept_switch_effect = {
	if = {
		limit = { NOT = { has_variable = mbq_kept_seen } }
		set_variable = { name = mbq_kept_seen value = yes }
		every_held_title = { # 2.1.46 (Maxim 25-10): jede gehaltene Grafschaft UND jede gehaltene Baronie einzeln (keine Vererbung mehr)
			limit = {
				OR = { tier = tier_county tier = tier_barony }
				NOT = { has_variable = mbq_gr_kept }
			}
			set_variable = { name = mbq_gr_kept value = yes }
		}
		set_variable = { name = mbq_kept_mig_2146 value = yes } # 2.1.46: ein neu markierter Spieler braucht die Umstellung nicht
	}
}


# 2.1.46 (Maxim 25-10): einmalige Umstellung je Spieler auf die Marke je Besitz. Scope = Spieler. Vorher schuetzte eine markierte Grafschaft ihre Baronien mit (ausser mbq_gr_free). Jetzt bekommt jede gehaltene Baronie
# (nicht der Sitz der Grafschaft, der steht fuer die Grafschaft) die eigene Marke, wenn sie bisher so geschuetzt war; danach fallen alle mbq_gr_free. Der Schutz aendert sich nicht.
# Verpachtete Titel (Tempel beim Kleriker) werden mitmarkiert, sie gehoeren dem Spieler weiter.
mbq_kept_migrate_2146_effect = {
	set_variable = { name = mbq_kept_mig_2146 value = yes }
	every_held_title = {
		limit = {
			tier = tier_barony
			is_capital_barony = no
			NOT = { has_variable = mbq_gr_kept }
			NOT = { has_variable = mbq_gr_free }
			de_jure_liege ?= { has_variable = mbq_gr_kept }
		}
		set_variable = { name = mbq_gr_kept value = yes }
	}
	every_held_title = {
		limit = { has_variable = mbq_gr_free }
		remove_variable = mbq_gr_free
	}
}

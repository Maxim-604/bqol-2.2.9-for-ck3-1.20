# 2.1.22 (Maxim 24-10): manueller Spielerwechsel (Konsole 'play <id>', Charakter-Wechsel) markiert die gehaltenen Grafschaften des neuen Charakters als Wertvoll, siehe mbq_kept_switch_effect.
# V common/on_action/player_change_on_actions.txt:3-6: on_player_character_change feuert bei ALLEN Wechseln, Root = neuer Spielercharakter, scope:previous_player_character = der alte.
# Erbfolge nach Tod ist auch ein Wechsel: dort ist der alte Charakter tot, darum verlangt der Trigger is_alive = yes (Tod/Erbfolge markiert nichts).
on_player_character_change = {
	on_actions = {
		mbq_player_switch_kept
	}
}

mbq_player_switch_kept = {
	trigger = {
		is_ai = no
		exists = scope:previous_player_character
		scope:previous_player_character = { is_alive = yes }
	}
	effect = {
		mbq_kept_switch_effect = yes
	}
}

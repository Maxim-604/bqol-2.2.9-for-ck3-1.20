# Better Quality of Life 2.0.0 (Demand Conversion): Schalter. Gesetzt = an bei mbq_dc_auto, mbq_dc_t*, mbq_dc_min_*; gesetzt = Haken weg bei mbq_dc_no_*.
mbq_dc_toggle_auto = {
	scope = character
	effect = {
		if = { limit = { has_global_variable = mbq_dc_auto } remove_global_variable = mbq_dc_auto }
		else = {
			set_global_variable = { name = mbq_dc_auto value = yes }
			if = { limit = { NOT = { has_variable = mbq_dc_armed } } mbq_dc_arm_effect = yes }
		}
	}
}

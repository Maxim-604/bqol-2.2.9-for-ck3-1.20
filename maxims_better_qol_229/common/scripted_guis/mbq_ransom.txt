mbq_rs_toggle_dungeon = {
	scope = character
	effect = {
		if = { limit = { has_global_variable = mbq_rs_no_dungeon } remove_global_variable = mbq_rs_no_dungeon }
		else = { set_global_variable = { name = mbq_rs_no_dungeon value = yes } }
		mbq_rs_preview_effect = yes
	}
}

mbq_rs_set_monthly = { # Fassung 4: Standard monatlich = keine Variable. Nachbesserung 13-10 (Test M5): Umstellung startet den Takt sofort neu (mbq_rs_arm_effect), alter Termin verfaellt
	scope = character
	effect = {
		if = {
			limit = { has_global_variable = mbq_rs_quarterly }
			remove_global_variable = mbq_rs_quarterly
			if = { limit = { has_global_variable = mbq_rs_auto } mbq_rs_arm_effect = yes }
		}
	}
}
mbq_rs_set_quarterly = { # alle drei Monate = Variable mbq_rs_quarterly (ersetzt mbq_rs_monthly)
	scope = character
	effect = {
		if = {
			limit = { NOT = { has_global_variable = mbq_rs_quarterly } }
			set_global_variable = { name = mbq_rs_quarterly value = yes }
			if = { limit = { has_global_variable = mbq_rs_auto } mbq_rs_arm_effect = yes }
		}
	}
}

	scope = character
	effect = {

	}
}
